\paragraph{Mutual self-attention.}
Two denoising processes run in parallel: the source process (real image inverted with DDIM, source prompt
$P_s=\varnothing$) and the target process with prompt $P$, started from the same noise $z_T$.
In the target process, the self-attention at step $t$ and layer $l$ is replaced by
\begin{equation}
\mathrm{EDIT}:=
\begin{cases}
\{Q,\;K_s,\;V_s\}, & t>S \text{ and } l>L,\\
\{Q,\;K,\;V\}, & \text{otherwise},
\end{cases}
\label{eq:masactrl}
\end{equation}

\paragraph{Mask-guided variant.}
With foreground masks $M_s, M$ obtained by thresholding the averaged $16\times16$ cross-attention map of the object token,
\begin{align}
f_o^l &= \mathrm{Attention}(Q^l, K_s^l, V_s^l; M_s), \qquad
f_b^l = \mathrm{Attention}(Q^l, K_s^l, V_s^l; 1-M_s),\\
\bar f^l &= f_o^l \odot M + f_b^l \odot (1-M).
\end{align}
